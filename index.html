strategy.ts 
import mangrove from "@mangrove/sdk";

const client = new mangrove.Client({
  apiKey: process.env.MANGROVE_API_KEY,
  network: "arbitrum",
});

// Define a strategy with hard risk bounds
const strategy = await client.strategies.create({
  name: "ETH Momentum",
  rules: [
    { condition: "rsi_4h > 60", action: "buy", size: "2%" },
    { condition: "price_change_1h < -5%", action: "sell_all" },
  ],
  riskControls: {
    maxDrawdown: 0.08,       // 8% hard limit
    maxPositionSize: 0.05,   // 5% per asset
    noLeverage: true,
  },
});

// Deploy on-chain — returns tx hash
const { txHash } = await strategy.deploy();
console.log(`Live at ${txHash}`);
